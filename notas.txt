#Construir el contenedor
docker build -t mi-grpc-app .

#Correr el contenedor
docker run -it -v $(pwd):/app mi-grpc-app bash
cd /app && mkdir -p build && cd build
cmake .. && make

#Compilación de los archivos proto
cd /app
protoc --grpc_out=src --cpp_out=src -I protos --plugin=protoc-gen-grpc=`which grpc_cpp_plugin` protos/suma.proto

#Con esto ya compilan todo
cd build
cmake -DCMAKE_EXPORT_COMPILE_COMMANDS=ON ..
make
